\begin{frame}{Project Goal}
  \begin{block}{Research contribution}
    MarketGyan introduces a Nepal-specific market-impact benchmark and model workflow for identifying whether news is relevant to NEPSE, mapping it to event types, sectors, symbols, and evidence, and supporting grounded post-market reporting.
  \end{block}
  \smallsep
  \begin{itemize}
    \item Focus: Nepal share-market language, entities, and event mechanisms.
    \item Dataset: human-adjudicated bilingual NEPSE-Impact-500.
    \item Model: XLM-R and FinBERT baselines; Qwen3-8B QLoRA structured adapter.
    \item Automation: RAG and deterministic market data remain the source of current facts.
  \end{itemize}
\end{frame}

\begin{frame}{Problem Reframing}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{Initial issue}
        General business news often looks financial but does not have a defensible NEPSE impact mechanism.
      \end{block}
      \begin{itemize}
        \item Product launches
        \item Commodity retail prices
        \item Foreign employment stories
        \item Non-listed company news
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Revised task}
        Learn market relevance, event taxonomy, symbol/sector mapping, and evidence-grounded potential impact.
      \end{block}
      \begin{itemize}
        \item Direct NEPSE impact
        \item Indirect sector or macro impact
        \item Hard-negative non-market stories
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

    \midrule
    XLM-R relevance & 0.707 & 0.665 & Best reliable relevance baseline; indirect class remains weak. \\
    XLM-R direction & 0.634 & 0.401 & Bearish and uncertain are usable; bullish and neutral need work. \\
    FinBERT English direction & 0.929 & 0.636 & Strong but only 14 English test rows, so it is a motivation baseline. \\

    \bottomrule
  \end{tabularx}

  \begin{block}{Interpretation}
    XLM-R is currently the reliable baseline. Qwen requires compact-output retraining before its result is meaningful.
  \end{block}

\begin{frame}{Error Audit Findings}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{block}{Baseline errors}
        \begin{itemize}
          \item XLM-R relevance: 22/75 errors.
          \item XLM-R direction: 15/41 errors.
          \item FinBERT: 1/14 errors.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Qwen errors}
        \begin{itemize}
          \item 44/75 invalid or truncated JSON outputs.
          \item 10 hard-negative false positives among valid JSON.
          \item Strong bias toward direct relevance.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \smallsep
  \begin{block}{Manual revalidation target}
    Prioritize hard negatives that both XLM-R and Qwen mark as relevant, plus bullish direct records predicted as uncertain.
  \end{block}
\end{frame}

\begin{frame}{Why Qwen Failed in First Run}
  \begin{itemize}
    \item Full gold JSON targets were too verbose for stable generation.
    \item Qwen was asked to generate summaries, rationales, evidence text, tags, and labels in one output.
    \item The chronological split produced a language/source shift: train was mostly English/direct while test had many Nepali hard negatives.
    \item The model learned output style but not a strong not-relevant gate.
  \end{itemize}
  \smallsep
  \begin{block}{Fix already implemented}
    Balanced duplicate-grouped split plus compact Qwen labels: canonical labels, sectors, symbols, confidence, and evidence sentence IDs only.
  \end{block}
\end{frame}

\begin{frame}{New Training Scheme}
  \begin{tabularx}{\textwidth}{@{}lX@{}}
    \toprule
    Part & Decision \\
    \midrule
    Split & Balanced duplicate-grouped 70/15/15; chronological retained as robustness check. \\
    Qwen target & Compact JSON only; no summary, rationale, evidence text, tags, or prose. \\
    Hard negatives & Oversample not-relevant, indirect, and Nepali hard-negative examples. \\
    Evaluation & JSON validity, grounding, relevance macro-F1, event macro-F1, direction macro-F1, sector/symbol F1. \\
    Report generation & Use RAG and deterministic data for current facts and citations. \\
    \bottomrule
  \end{tabularx}
\end{frame}
